41def fingerprint(config: Config, preflight_output: str) -> str: 

42 """A hash of `check`, `preflight`, `require_env`, `require_up_to_date` and the standard output 

43 of the preflight command (the tool versions it reports); the other keys do not change what a 

44 pass means. JSON keeps values that run together in a join apart.""" 

45 values = [ 

46 config.check, 

47 config.preflight, 

48 list(config.require_env), 

49 config.require_up_to_date, 

50 preflight_output, 

51 ] 

52 return hashlib.sha256(json.dumps(values).encode("utf-8")).hexdigest()[:LENGTH]

60def result_message(sha: str, facts: str | None = None) -> list[str]: 

61 """The paragraphs of the message of the commit a result ref points at: they name the commit 

62 that was checked, and then hold the facts of the run (see `run_facts.encode`) when there are 

63 any.""" 

64 paragraphs = [SUBJECT, f"{NAMES}{sha}"] 

65 return paragraphs if facts is None else [*paragraphs, f"{FACTS}{facts}"]

68def passed_commit(log: str, key: Key) -> str | None: 

69 """The commit a result ref names, read from `git log -1 --format=%T%n%b` of the ref; None 

70 when the ref points at a commit of another tree or its message names no commit. The commit is 

71 the first line of the body: a paragraph of facts may follow.""" 

72 tree, _, body = log.partition("\n") 

73 first = body.strip().partition("\n")[0] 

74 if tree != key.tree or not first.startswith(NAMES): 

75 return None 

76 return first.removeprefix(NAMES)

79def passed_facts(log: str, key: Key) -> RunFacts | None: 

80 """The facts of the run a result ref records, read from the same log as `passed_commit`; None 

81 when the pass has none or they cannot be read, which leaves it a pass.""" 

82 if passed_commit(log, key) is None: 

83 return None 

84 paragraphs = log.partition("\n")[2].strip().split("\n\n") 

85 if len(paragraphs) < 2 or not paragraphs[1].startswith(FACTS): 

86 return None 

87 return from_line(paragraphs[1].removeprefix(FACTS))

90def pushes_to_origin(args: Sequence[str], env: Mapping[str, str]) -> bool: 

91 """Whether the push the hook checks goes to the remote named `origin`: git passes the name 

92 (or the url, for `git push <url>`) as the first argument, pre-commit passes none and sets 

93 `PRE_COMMIT_REMOTE_NAME`. A hook that is told neither (lefthook) does not know, which is no.""" 

94 name = args[0] if args else env.get("PRE_COMMIT_REMOTE_NAME") 

95 return name == REMOTE

98def expired_refs(listing: str, now: int) -> list[str]: 

99 """The refs older than 30 days at `now` (a Unix time), from the output of 

100 `git for-each-ref --format='%(refname) %(committerdate:unix)'`. A ref dated in the future or 

101 without a readable date is kept: the tool removes only what it can age.""" 

102 expired: list[str] = [] 

103 for line in listing.splitlines(): 

104 ref, _, stamp = line.rpartition(" ") 

105 if ref and stamp.isdigit() and now - int(stamp) > EXPIRES_AFTER: 

106 expired.append(ref) 

107 return expired

151def _cause(git_output: str) -> str: 

152 """Git's wording is not an interface, so only a status and the `[remote rejected]` line are 

153 read; anything else claims nothing about why. A `remote:` line is the remote author's text, 

154 not the transport's, so no status is read from it. Lines end at a line feed only: git prints the 

155 remote's text on one line, and `splitlines` would cut it at a vertical tab, a form feed or U+2028 in it.""" 

156 own_lines = "\n".join( 

157 line for line in git_output.split("\n") if not line.startswith("remote:") 

158 ) 

159 for match in _STATUS.finditer(own_lines): 

160 if 400 <= int(match["status"]) <= 599: 

161 return f"{REMOTE} refused the ref, HTTP {match['status']}" 

162 if "[remote rejected]" in git_output: 

163 return f"{REMOTE} rejected the ref" 

164 return "git could not push it"
