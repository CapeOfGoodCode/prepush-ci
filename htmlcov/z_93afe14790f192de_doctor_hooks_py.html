10_ACTIVATION = { 

11 "pre-commit": "pre-commit install",

78def native_check(hook: str, facts: NativeFacts) -> Check: 

79 """Whether git would run our script for `hook`: right directory, executable, a live interpreter.""" 

80 if not facts.ours: 

81 if facts.configured is None: 

82 text = f"core.hooksPath is not set, so git does not run {facts.directory}" 

83 else: 

84 text = ( 

85 f"core.hooksPath is {facts.configured}, not {facts.directory}: " 

86 "another tool took over the hooks" 

87 ) 

88 return fail(hook, text, _REINSTALL) 

89 script = facts.scripts[hook] 

90 if not script.exists: 

91 return fail(hook, f"{facts.directory}/{hook} does not exist", _REINSTALL) 

92 if not script.executable: 

93 text = f"{facts.directory}/{hook} is not executable, so git skips it" 

94 return fail(hook, text, _REINSTALL) 

95 if facts.interpreter is None: 

96 text = f"the file python is missing from {facts.directory}" 

97 return fail(hook, text, _REINSTALL) 

98 if not facts.interpreter_exists: 

99 text = f"the interpreter {facts.interpreter} no longer exists" 

100 return fail(hook, text, _REINSTALL) 

101 return ok(hook, f"{facts.directory}/{hook} runs {facts.interpreter}")

104def manager_check( 

105 hook: str, facts: ManagerFacts, own: str | None, binary: Callable[[], Binary] 

106) -> Check: 

107 """Whether the manager's file has our block for `hook`, git would run the manager, and the 

108 manager finds the `prepush-ci` whose `--version` line is `own` (None when doctor cannot read 

109 its own). `binary` looks it up and runs it; it is called only for a check that has passed 

110 everything else.""" 

111 problem = _manager_problem(hook, facts) 

112 if problem is None: 

113 if own is None: 

114 set_up = f"{facts.file} has the block and {facts.manager} is activated" 

115 text = f"{set_up}, but this prepush-ci cannot read its own version, so the hook's cannot be compared" 

116 return fail( 

117 hook, 

118 text, 

119 "run doctor from an installed prepush-ci (`uv tool install prepush-ci`)", 

120 ) 

121 return _binary_check(hook, facts, own, binary()) 

122 text, fix = problem 

123 if facts.detected: 

124 fix = f"{fix}; {_NATIVE_HINT}" 

125 return fail(hook, text, fix)

128def _binary_check(hook: str, facts: ManagerFacts, own: str, found: Binary) -> Check: 

129 set_up = f"{facts.file} has the block and {facts.manager} is activated" 

130 if found.path is None: 

131 text = f"{set_up}, but no prepush-ci is on PATH" 

132 return fail(hook, text, f"put the prepush-ci that prints `{own}` on PATH") 

133 if found.line is None: 

134 text = ( 

135 f"{set_up}, but the version of {found.path} could not be read: {found.why}" 

136 ) 

137 return fail(hook, text, f"put the prepush-ci that prints `{own}` first on PATH") 

138 if found.line != own: 

139 text = f"{set_up}, but the hook runs {found.path}: {found.line}, not {own}" 

140 return fail(hook, text, f"put the prepush-ci that prints `{own}` first on PATH") 

141 return ok(hook, f"{set_up}; the hook runs {found.path}: {found.line}")

144def _manager_problem(hook: str, facts: ManagerFacts) -> tuple[str, str] | None: 

145 """What is wrong and how to fix it; None when nothing is.""" 

146 manager = facts.manager 

147 install = ( 

148 "run `prepush-ci install`" 

149 if facts.detected 

150 else f"run `prepush-ci install --hooks {manager}`" 

151 ) 

152 if facts.content is None: 

153 return f"{facts.file} does not exist", install 

154 lines = split_lines(facts.content)[0] 

155 try: 

156 block = find_block(lines, f"prepush-ci hook {hook}") 

157 except Refusal as refusal: 

158 return f"{facts.file}: {refusal}", install 

159 if block is None: 

160 return f"{facts.file} has no prepush-ci block for {hook}", install 

161 if manager == "husky": 

162 return _husky_problem(hook, facts) 

163 hidden = not _shows_messages(lines[block[0] + 1 : block[1]]) 

164 if manager == "pre-commit" and hook == "pre-push" and hidden: 

165 text = f"{facts.file}: the {PRE_PUSH_ID} hook has no `verbose: true`, so pre-commit hides what it prints" 

166 return text, install 

167 return _activation_problem(hook, facts)

170def _shows_messages(block: Sequence[str]) -> bool: 

171 """Whether the `prepush-ci-pre-push` hook of a pre-commit block holds the line `verbose: true`: its 

172 lines run from its `- id:` line to the next one.""" 

173 inside = False 

174 for line in block: 

175 text = line.strip() 

176 if text.startswith("- id:"): 

177 inside = text.removeprefix("- id:").strip() == PRE_PUSH_ID 

178 elif inside and text == "verbose: true": 

179 return True 

180 return False

183def _husky_problem(hook: str, facts: ManagerFacts) -> tuple[str, str] | None: 

184 fix = f"run `{_ACTIVATION['husky']}`" 

185 if facts.hooks_path not in HUSKY_PATHS: 

186 shown = facts.hooks_path or "unset" 

187 text = f"husky is not set up on this clone: core.hooksPath is {shown}, not .husky/_" 

188 return text, fix 

189 where = f"{facts.hooks_path}/{hook}" 

190 if not facts.installed.exists: 

191 return f"husky: {where} does not exist, so git runs no hook", fix 

192 if not facts.installed.executable: 

193 return f"husky: {where} is not executable, so git skips it", fix 

194 return None

197def _inactive_reason(manager: str, script: Script) -> str | None: 

198 """Why git would not run `manager` from `script`; None when it would.""" 

199 if not script.exists: 

200 return "does not exist" 

201 if not script.executable: 

202 return "is not executable" 

203 if _NAMED_IN_SCRIPT[manager] not in script.content: 

204 return f"does not run {manager}" 

205 return None

208def hooks_active(manager: str, scripts: Sequence[Script]) -> bool: 

209 """Whether every one of the hook `scripts` is activated for `manager`: the test of the 

210 activation check, which `install` uses to know whether to print the activation step.""" 

211 return all(_inactive_reason(manager, script) is None for script in scripts)

214def _activation_problem(hook: str, facts: ManagerFacts) -> tuple[str, str] | None: 

215 manager = facts.manager 

216 reason = _inactive_reason(manager, facts.installed) 

217 if reason is None: 

218 return None 

219 return ( 

220 f"{manager} is not activated: {facts.hooks_directory}/{hook} {reason}", 

221 f"run `{_ACTIVATION[manager]}`",
