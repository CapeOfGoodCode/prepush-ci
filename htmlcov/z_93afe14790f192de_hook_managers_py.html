15def split_lines(text: str) -> tuple[list[str], str]: 

16 """The lines of `text` without their line ends, and the line end the file uses.""" 

17 eol = "\r\n" if "\r\n" in text else "\n" 

18 lines = text.split(eol) 

19 if lines[-1] == "": # the final line end is not a line 

20 lines.pop() 

21 return lines, eol

24def join_lines(lines: Sequence[str], eol: str) -> str: 

25 """The text of `lines`, every line ended.""" 

26 return "".join(line + eol for line in lines)

29def find_block(lines: Sequence[str], needle: str = "") -> tuple[int, int] | None: 

30 """The indexes of the begin and end lines of the first block whose body contains `needle`.""" 

31 start = 0 

32 while start < len(lines): 

33 if lines[start].strip() != BEGIN: 

34 start += 1 

35 continue 

36 end = next( 

37 (i for i in range(start + 1, len(lines)) if lines[i].strip() == END), None 

38 ) 

39 if end is None: 

40 raise Refusal(f"it has `{BEGIN}` without a matching `{END}`") 

41 if needle in "\n".join(lines[start + 1 : end]): 

42 return start, end 

43 start = end + 1 

44 return None

47def place_block( 

48 lines: Sequence[str], block: Sequence[str], at: int, needle: str = "" 

49) -> list[str]: 

50 """`lines` with `block` (begin and end lines included) replacing the block that holds 

51 `needle`, or inserted before line `at` when there is none.""" 

52 span = find_block(lines, needle) 

53 if span is None: 

54 span = (at, at - 1) 

55 return [*lines[: span[0]], *block, *lines[span[1] + 1 :]]

58# What an entry does when the tool is not on PATH (spec `install`): `commit-msg` warns and lets 

59# the commit go on, `pre-push` refuses, as the native hooks do. The entries are the only code 

60# that runs then, so the guard is written into each of them (`command -v` is POSIX sh). 

61_NOT_FOUND = "prepush-ci is not on PATH, so" 

62_WARNING = f"{_NOT_FOUND} no trailer is written - install it" 

63_REFUSAL = f"{_NOT_FOUND} this push cannot be checked - install it" 

64_FOUND = "if command -v prepush-ci >/dev/null 2>&1; then" 

65 

66PRE_COMMIT_FILE = ".pre-commit-config.yaml" 

67_TYPES_KEY = "default_install_hook_types" 

68_STAGES_KEY = "default_stages" 

69_WANTED_TYPES = ("commit-msg", "pre-push") 

70_BLOCK_ITEM = re.compile(r"\s*-\s+(\S+)\s*(#.*)?") 

71_FLOW_TYPES = re.compile(rf"{_TYPES_KEY}:\s*\[([^\]]*)\]\s*(#.*)?")

76 return next((i for i, line in enumerate(lines) if line.startswith(f"{key}:")), None)

92def _pre_commit_entry(hook: str) -> str: 

93 """The `entry:` of `hook`: `sh -c` with the guard; pre-commit appends the message file, which 

94 `sh` hands to the script as `$1`.""" 

95 if hook == "commit-msg": 

96 script = f'{_FOUND} exec prepush-ci hook commit-msg "$1"; else echo "{_WARNING}" >&2; fi' 

97 else: 

98 script = f'{_FOUND} exec prepush-ci hook pre-push; else echo "{_REFUSAL}" >&2; exit 1; fi' 

99 return f"sh -c '{script}' prepush-ci"

102def pre_commit_block(indent: str) -> list[str]: 

103 """The `local` repo with the two hooks, its list item at `indent`.""" 

104 lines = [BEGIN, "- repo: local", " hooks:"] 

105 for hook, extra in ( 

106 # pre-commit hides the output of a hook that passes, and with it the warning 

107 ("commit-msg", ["stages: [commit-msg]", "verbose: true"]), 

108 ( 

109 "pre-push", 

110 ["stages: [pre-push]", "pass_filenames: false", "always_run: true"], 

111 ), 

112 ): 

113 lines += [ 

114 f" - id: prepush-ci-{hook}", 

115 f" name: prepush-ci {hook}", 

116 f" entry: {_pre_commit_entry(hook)}", 

117 " language: system", 

118 *(f" {line}" for line in extra), 

119 ] 

120 return [f"{indent}{line}" for line in [*lines, END]]

123def edit_pre_commit(text: str) -> str: 

124 """The `.pre-commit-config.yaml` with the block at the end of `repos:` and both hook types 

125 installed, the hooks it already had kept at the stages they ran at.""" 

126 lines, eol = split_lines(text) 

127 repos = _find(lines, "repos") 

128 if repos is None or not _blank_or_comment(lines[repos][len("repos:") :]): 

129 raise Refusal("`repos` must be a block list: `repos:` and `- repo:` entries") 

130 indent, last = None, repos 

131 for i in range(repos + 1, len(lines)): 

132 line = lines[i] 

133 if _blank_or_comment(line): 

134 continue 

135 if _indent(line) == 0 and not line.startswith("-"): 

136 break # the next top-level key 

137 if indent is None: 

138 if not line.lstrip().startswith("-"): 

139 raise Refusal("`repos` must be a block list of `- repo:` entries") 

140 indent = _indent(line) 

141 last = i 

142 if indent is None: 

143 raise Refusal("`repos` has no entries: add a `- repo:` or remove the key") 

144 lines = place_block(lines, pre_commit_block(" " * indent), last + 1) 

145 before = _with_hook_types(lines) 

146 if before is not None and _find(lines, _STAGES_KEY) is None: 

147 repos = _find(lines, "repos") 

148 assert repos is not None # still there: only lines above and below it changed 

149 lines.insert(repos, f"{_STAGES_KEY}: [{', '.join(before)}]") 

150 return join_lines(lines, eol)

153def _with_hook_types(lines: list[str]) -> list[str] | None: 

154 """Make `default_install_hook_types` in `lines` hold `commit-msg` and `pre-push`. The types 

155 it held before when any was added, else None.""" 

156 key = _find(lines, _TYPES_KEY) 

157 if key is None: 

158 repos = _find(lines, "repos") 

159 assert repos is not None # the caller found it 

160 lines.insert(repos, f"{_TYPES_KEY}: [pre-commit, commit-msg, pre-push]") 

161 return ["pre-commit"] # what pre-commit installs when the key is absent 

162 flow = _FLOW_TYPES.fullmatch(lines[key]) 

163 if flow is not None: 

164 items = [item.strip() for item in flow[1].split(",") if item.strip()] 

165 missing = _missing_types(items) 

166 if missing: 

167 comment = f" {flow[2]}" if flow[2] else "" 

168 lines[key] = f"{_TYPES_KEY}: [{', '.join([*items, *missing])}]{comment}" 

169 return _bare(items) 

170 return None 

171 if not _blank_or_comment(lines[key][len(_TYPES_KEY) + 1 :]): 

172 raise Refusal( 

173 f"`{_TYPES_KEY}` must be a list on one line or a block list of hook types" 

174 ) 

175 last, items = _block_items(lines, key) 

176 missing = _missing_types(items) 

177 if not missing: 

178 return None 

179 pad = " " * _indent(lines[last]) 

180 lines[last + 1 : last + 1] = [f"{pad}- {t}" for t in missing] 

181 return _bare(items)

184def _missing_types(items: Sequence[str]) -> list[str]: 

185 present = _bare(items) 

186 return [t for t in _WANTED_TYPES if t not in present]

189def _block_items(lines: Sequence[str], key: int) -> tuple[int, list[str]]: 

190 """The index of the last item and the items of the block list under line `key`.""" 

191 last, items = key, [] 

192 for i in range(key + 1, len(lines)): 

193 if _blank_or_comment(lines[i]): 

194 continue 

195 item = _BLOCK_ITEM.fullmatch(lines[i]) 

196 if item is None: 

197 break 

198 last = i 

199 items.append(item[1]) 

200 if not items: 

201 raise Refusal(f"`{_TYPES_KEY}` has no items: list the hook types") 

202 return last, items

205HUSKY_DIRECTORY = ".husky" 

206_HOOKS = ("commit-msg", "pre-push") 

207_HUSKY_SHEBANG = "#!/usr/bin/env sh" 

208# The pre-push block reads git's standard input once, hands it to the tool, ends the hook with 

209# its status when it fails, and gives the same input to the lines after it. 

210_HUSKY_BLOCKS = { 

211 "commit-msg": [ 

212 BEGIN, 

213 _FOUND, 

214 ' prepush-ci hook commit-msg "$1" || exit $?', 

215 "else",

217 "fi", 

218 END,

227 " exit 1", 

228 "fi", 

229 "exec <<PREPUSH_CI_REFS", 

230 "$refs", 

231 "PREPUSH_CI_REFS", 

232 END, 

233 ],

236LEFTHOOK_FILES = ("lefthook.yml", "lefthook.yaml", ".lefthook.yml", ".lefthook.yaml") 

237_LEFTHOOK_RUN = { 

238 "commit-msg": [

241 "pre-push": [

260def choose(hooks: str | None, found: Sequence[str]) -> str: 

261 """The hooks to install: the one asked for, else the manager found, else native.""" 

262 if hooks is not None: 

263 return hooks 

264 if len(found) > 1: 

265 raise Refusal( 

266 f"several hook managers found ({', '.join(found)}): " 

267 f"say which with --hooks {{{'|'.join(CHOICES)}}}" 

268 ) 

269 return found[0] if found else "native"

277def manager_files(manager: str, present: Sequence[str]) -> list[str]: 

278 """The files `install` edits for `manager`, relative to the worktree root; `present` are the 

279 names from `MARKERS` that exist."""

287def edit_file(manager: str, name: str, text: str | None) -> str: 

288 """The new contents of file `name` of `manager`, `text` being its contents or None.""" 

289 if manager == "husky": 

290 return edit_husky(name.rsplit("/", 1)[-1], text) 

291 if manager == "lefthook":

293 return edit_pre_commit(text or "")

302def edit_lefthook(text: str) -> str: 

303 """The lefthook config with the command in `commit-msg` and `pre-push`, each added to the 

304 `commands:` it has or to a hook section added at the end.""" 

305 lines, eol = split_lines(text) 

306 for hook in _HOOKS: 

307 lines = _lefthook_hook(lines, hook) 

308 return join_lines(lines, eol)

311def _lefthook_hook(lines: list[str], hook: str) -> list[str]: 

312 key = _find(lines, hook) 

313 if key is None: 

314 return [*lines, f"{hook}:", " commands:", *lefthook_block(hook, " ")] 

315 where = f"`{hook}` must be a mapping with a block-style `commands:`" 

316 if not _blank_or_comment(lines[key][len(hook) + 1 :]): 

317 raise Refusal(where) 

318 end = next( 

319 ( 

320 i 

321 for i in range(key + 1, len(lines)) 

322 if not _blank_or_comment(lines[i]) and _indent(lines[i]) == 0 

323 ), 

324 len(lines), 

325 ) 

326 commands = next( 

327 (i for i in range(key + 1, end) if _COMMANDS.fullmatch(lines[i])), None 

328 ) 

329 if commands is None: 

330 raise Refusal(where) 

331 entries = next( 

332 ( 

333 _indent(lines[i]) 

334 for i in range(commands + 1, end) 

335 if not _blank_or_comment(lines[i]) 

336 ), 

337 0, 

338 ) 

339 inner = ( 

340 entries if entries > _indent(lines[commands]) else _indent(lines[commands]) + 2 

341 ) 

342 return place_block( 

343 lines, lefthook_block(hook, " " * inner), commands + 1, f"hook {hook}" 

344 )

347def edit_husky(hook: str, text: str | None) -> str: 

348 """A husky hook file with the block before every other line (after a `#!` line): the whole 

349 file when `text` is None.""" 

350 if text is None: 

351 return join_lines([_HUSKY_SHEBANG, *_HUSKY_BLOCKS[hook]], "\n") 

352 lines, eol = split_lines(text) 

353 after_shebang = 1 if lines and lines[0].startswith("#!") else 0 

354 return join_lines(place_block(lines, _HUSKY_BLOCKS[hook], after_shebang), eol)
